function carregarAnimaisDoLocalStorage() {
  const raw = localStorage.getItem(STORAGE_KEY);
  if (!raw) return [];
  const data = JSON.parse(raw);
  return data.animais || data || [];
}

    function salvarAnimaisNoLocalStorage(lista) {
  const payload = {
    animais: lista,
    version: '2.8-localStorage',
    savedAt: new Date().toISOString()
  };
  localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
  localStorage.setItem('cofre_animais', JSON.stringify(lista)); // backup extra
}

function cadastrarAnimal(novo) {
  animais.push(novo);
  salvarAnimaisNoLocalStorage(animais); // <- SALVA AQUI
}

function excluirAnimal(id) {
  animais = animais.filter(a => a.id !== id);
  salvarAnimaisNoLocalStorage(animais); // <- SALVA AQUI TAMBÉM
}
